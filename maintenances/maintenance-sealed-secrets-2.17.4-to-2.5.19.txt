### Sealed Secrets Maintenance: 2.17.4 to 2.5.19

### Notes

# (opt) refers to /opt, the standard install path for Genestack.
# (etc) refers to /etc, the standard override path for Genestack.
# Replace all placeholder values in angle brackets before using this runbook.
# Keep this runbook operational. Prefer exact commands, exact file paths, expected outputs, and explicit validation steps.

### Validation

# Kubernetes 1.35.4 satisfies sealed-secrets chart 2.5.x requirements.
# The chart version jumps from 2.17.4 to 2.5.19 — this is a version scheme change, NOT a downgrade.
# Bitnami re-based chart versioning; the app version moves from 0.27.x to 0.31.0.
# Validated source version: 2.17.4.
# Validated target version: 2.5.19.

# Major operational risks:
# - Existing SealedSecrets must remain decryptable after the operator upgrade.
# - The encryption key in the sealed-secrets-key secret must be preserved.
# - The kubeseal CLI version should be compatible with the new controller.
# - If the controller image repository changes, verify the new registry is accessible.

# Read upstream upgrade guide:
#   https://github.com/bitnami-labs/sealed-secrets
#   https://github.com/bitnami/charts/tree/main/bitnami/sealed-secrets

### Goal

# Upgrade sealed-secrets from chart 2.17.4 to chart 2.5.19 without breaking
# existing SealedSecret decryption, with the controller and webhook healthy.

### Prep

## Deployment Node

# Identify the deployment host with kubectl, helm access.

# Verify current health:
kubectl -n sealed-secrets get pods

# Expected:
# - sealed-secrets controller: Running
# - SealedSecret status controller: Running

# Verify current chart version:
helm list -n sealed-secrets -f sealed-secrets
grep '^[[:space:]]*sealed-secrets:' /etc/genestack/helm-chart-versions.yaml

# Expected:
# - sealed-secrets: 2.17.4

# Verify the encryption key is present:
kubectl -n sealed-secrets get secret sealed-secrets-key -o yaml > /tmp/sealed-secrets-key-backup.yaml
kubectl -n sealed-secrets get secret sealed-secrets-key -o jsonpath='{.data.tls\.crt}' | base64 -d | wc -c

# Expected: non-zero byte count (the key exists)

# Verify kubeseal CLI version (if installed):
kubeseal --version 2>/dev/null || echo "kubeseal not installed"

# Backup all SealedSecrets:
kubectl get sealedsecrets -A -o yaml > /tmp/sealedsecrets-all.yaml
helm -n sealed-secrets get values sealed-secrets > /tmp/sealed-secrets-helm-values.yaml

# Test that an existing SealedSecret can be decrypted (spot check):
# Pick one SealedSecret and verify its encrypted data:
kubectl get sealedsecrets -A -o jsonpath='{.items[0].metadata.name}' | head -1

# Verify cluster health:
kubectl get nodes
kubectl get pods -A --no-headers | awk '$4 != "Running" && $4 != "Completed" {print}'

## Configuration Review

# Check the existing override file:
ls base-helm-configs/sealed-secrets/

# Verify the override file renders with the new chart version:
helm template sealed-secrets bitnami/sealed-secrets \
  --version 2.5.19 \
  -f /opt/genestack/base-helm-configs/sealed-secrets/helm-sealed-secrets-overrides.yaml \
  --namespace sealed-secrets > /tmp/sealed-secrets-render.yaml 2>&1

grep -i "error\|fail\|warning" /tmp/sealed-secrets-render.yaml | head -10 || echo "No errors found"

## Pre-Change Safety Checks

# Check for unhealthy sealed-secrets pods:
kubectl -n sealed-secrets get pods

# Check for SealedSecret reconciliation errors:
kubectl get sealedsecrets -A -o json | jq -r '.items[] | .metadata.namespace + "/" + .metadata.name + " - " + (.status.conditions[]? | select(.type=="ailed") | .message)' 2>/dev/null || true

# Verify all namespaces with SealedSecrets have the correct controller running:
for ns in $(kubectl get sealedsecrets -A -o jsonpath='{.items[*].metadata.namespace}' | tr ' ' '\n' | sort -u); do
  echo "=== $ns ==="
  kubectl get sealedsecret -n "$ns" --no-headers | wc -l
done

# Disable HPA during upgrade:
cd /opt/genestack
/opt/genestack/scripts/manage_hpa.sh lock 2>/dev/null || true

### Execute

## Update the Target Version

# Edit /etc/genestack/helm-chart-versions.yaml:
# sealed-secrets: 2.5.19

## Run a Dry-Run

# Test that the chart renders correctly:
helm template sealed-secrets bitnami/sealed-secrets \
  --version 2.5.19 \
  --namespace sealed-secrets \
  -f /opt/genestack/base-helm-configs/sealed-secrets/helm-sealed-secrets-overrides.yaml > /tmp/sealed-secrets-render.yaml 2>&1

# Check for errors:
grep -i "error\|fail\|warning" /tmp/sealed-secrets-render.yaml | head -20 || echo "No errors found"

## Run the Helm Upgrade

cd /opt/genestack
./bin/install-sealed-secrets.sh

# Wait for the rollout:
kubectl -n sealed-secrets rollout status deployment/sealed-secrets -c sealed-secrets --timeout=300s 2>/dev/null || \
  kubectl -n sealed-secrets rollout status deployment/sealed-secrets --timeout=300s

# Wait for webhook:
kubectl -n sealed-secrets rollout status deployment/sealed-secrets -c sealed-secrets-keymaker --timeout=300s 2>/dev/null || true

### Post-Maint

# Verify deployed version:
grep '^[[:space:]]*sealed-secrets:' /etc/genestack/helm-chart-versions.yaml
helm -n sealed-secrets list -f sealed-secrets
kubectl -n sealed-secrets get deployment sealed-secrets -o jsonpath='{range .spec.template.spec.containers[*]}{.image}{"\n"}{end}'

# Verify the encryption key is still present:
kubectl -n sealed-secrets get secret sealed-secrets-key -o yaml > /tmp/sealed-secrets-key-after.yaml
# Compare with /tmp/sealed-secrets-key-backup.yaml to ensure the same key
diff /tmp/sealed-secrets-key-backup.yaml /tmp/sealed-secrets-key-after.yaml || echo "Key differs (WARNING: investigate if SealedSecrets fail to decrypt)"

# Verify all pods are Running:
kubectl -n sealed-secrets get pods

# Verify existing SealedSecrets are still valid:
# The controller should reconcile them without issues:
kubectl get sealedsecrets -A --no-headers | wc -l
kubectl get sealedsecrets -A -o json | jq -r '.items[] | select(.status.conditions[]? | select(.type=="Ready") | .status != "True") | .metadata.namespace + "/" + .metadata.name + " - NOT READY"' 2>/dev/null

# If all SealedSecrets show Ready, no output means all are healthy.

# Update kubeseal CLI (if installed):
# Download the matching kubeseal version:
#   wget https://github.com/bitnami-labs/sealed-secrets/releases/download/v0.31.0/kubeseal-0.31.0-linux-amd64.tar.gz -O kubeseal.tar.gz
#   sudo tar -xzf kubeseal.tar.gz -C /usr/local/bin kubeseal

# Verify sealed-secrets controller is running the correct version:
kubectl -n sealed-secrets get deployment sealed-secrets -o yaml | grep "image:" | head -5

# Re-enable HPA:
cd /opt/genestack
/opt/genestack/scripts/manage_hpa.sh unlock 2>/dev/null || true

# Clean up:
rm -f /tmp/sealed-secrets-render.yaml

### Troubleshooting

## Common Failure Signals

# If existing SealedSecrets fail to decrypt:
# 1. Verify the encryption key was preserved:
diff /tmp/sealed-secrets-key-backup.yaml /tmp/sealed-secrets-key-after.yaml

# 2. If the key was lost, restore it:
kubectl apply -f /tmp/sealed-secrets-key-backup.yaml

# 3. Check controller logs:
kubectl -n sealed-secrets logs deployment/sealed-secrets --tail=100

# If the controller image is incorrect:
kubectl -n sealed-secrets get deployment sealed-secrets -o yaml | grep "image:"

# If new SealedSecrets cannot be created:
# Verify webhook configuration:
kubectl get validatingwebhookconfiguration sealed-secrets --no-headers 2>/dev/null || echo "Webhook not found"
kubectl get mutatingwebhookconfiguration sealed-secrets --no-headers 2>/dev/null || echo "Mutating webhook not found"

# Verify kubeseal can connect to the controller:
# kubectl -n sealed-secrets port-forward svc/sealed-secrets 8080 &
# kubeseal --format yaml --controller-namespace=sealed-secrets < /tmp/test-secret.yaml

## Rollback

# If the upgrade fails:
helm -n sealed-secrets rollback sealed-secrets <revision>

# Verify the key is intact:
kubectl -n sealed-secrets get secret sealed-secrets-key -o yaml

# Re-enable HPA:
cd /opt/genestack
/opt/genestack/scripts/manage_hpa.sh unlock 2>/dev/null || true

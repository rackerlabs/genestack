### Redis-Sentinel Maintenance: 0.16.12 to 0.16.13

### Notes

# (opt) refers to /opt, the standard install path for Genestack.
# (etc) refers to /etc, the standard override path for Genestack.
# Replace all placeholder values in angle brackets before using this runbook.
# Keep this runbook operational. Prefer exact commands, exact file paths, expected outputs, and explicit validation steps.

### Validation

# Kubernetes 1.35.4 satisfies redis-sentinel chart 0.16.x requirements.
# Upgrading from 0.16.12 to 0.16.13 is a patch version bump.
# The OT-helm redis-sentinel chart uses Bitnami redis images.
# Validated source version: 0.16.12.
# Validated target version: 0.16.13.

# Major operational risks:
# - Minimal — patch version bump.
# - Redis Sentinel data is in PVCs — should be preserved.
# - The sentinel cluster topology must remain stable.

# Read upstream upgrade guide:
#   https://bitnami.com/stack/redis-sentinel

### Goal

# Upgrade redis-sentinel from 0.16.12 to 0.16.13 without Sentinel quorum disruption.

### Prep

## Deployment Node

# Verify current health:
kubectl -n redis-system get pods | grep sentinel

# Verify current version:
helm list -n redis-system -f redis-sentinel
grep '^[[:space:]]*redis-sentinel:' /etc/genestack/helm-chart-versions.yaml

# Expected:
# - redis-sentinel: 0.16.12

# Backup Sentinel state:
helm -n redis-system get values redis-sentinel > /tmp/redis-sentinel-helm-values.yaml
kubectl -n redis-system get statefulset redis-sentinel -o yaml > /tmp/redis-sentinel-statefulset.yaml

# Verify Sentinel quorum (should be 3 or more pods, majority available):
kubectl -n redis-system get statefulset redis-sentinel -o jsonpath='{.spec.replicas}'
kubectl -n redis-system get pods -l app=redis-sentinel

# Verify cluster health:
kubectl get nodes
kubectl get pods -A --no-headers | awk '$4 != "Running" && $4 != "Completed" {print}'

## Pre-Change Safety Checks

# Verify Sentinel health:
for pod in $(kubectl -n redis-system get pods -l app=redis-sentinel -o name); do
  echo "=== $pod ==="
  kubectl -n redis-system exec "$pod" -- redis-cli -p 26379 ping 2>/dev/null || echo "Failed to ping sentinel"
done

# Verify master is known to all sentinels:
for pod in $(kubectl -n redis-system get pods -l app=redis-sentinel -o name); do
  echo "=== $pod ==="
  kubectl -n redis-system exec "$pod" -- redis-cli -p 26379 SENTINEL master mymaster 2>/dev/null | head -5
done

### Execute

## Update the Target Version

# Edit /etc/genestack/helm-chart-versions.yaml:
# redis-sentinel: 0.16.13

## Run the Helm Upgrade

cd /opt/genestack
helm repo add ot-helm https://ot-container-kit.github.io/helm-charts
helm repo update
helm upgrade --install redis-sentinel ot-helm/redis-sentinel \
  --version 0.16.13 \
  --namespace=redis-system \
  --timeout 300m \
  -f /opt/genestack/base-helm-configs/redis-sentinel/redis-sentinel-helm-overrides.yaml

# Wait for rollout:
kubectl -n redis-system rollout status statefulset/redis-sentinel --timeout=600s

### Post-Maint

# Verify deployed version:
grep '^[[:space:]]*redis-sentinel:' /etc/genestack/helm-chart-versions.yaml
helm -n redis-system list -f redis-sentinel

# Verify all Sentinel pods are Running:
kubectl -n redis-system get pods -l app=redis-sentinel

# Verify Sentinel quorum:
for pod in $(kubectl -n redis-system get pods -l app=redis-sentinel -o name); do
  echo "=== $pod ==="
  kubectl -n redis-system exec "$pod" -- redis-cli -p 26379 ping 2>/dev/null || echo "Failed to ping sentinel"
done

# Verify master is known:
for pod in $(kubectl -n redis-system get pods -l app=redis-sentinel -o name | head -1); do
  kubectl -n redis-system exec "$pod" -- redis-cli -p 26379 SENTINEL master mymaster 2>/dev/null | head -5
done

### Troubleshooting

## Rollback
helm -n redis-system rollback redis-sentinel <revision>
